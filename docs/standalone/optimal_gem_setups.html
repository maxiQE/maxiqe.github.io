def filter_data(data: list[dict[str, ...]], query: dict[str, ...]):
    filtered = [
        elem
        for elem in data
        if (
            elem["n_sets"] == query["n_sets"]
            and elem["overcap"] >= query["overcap"]
            and all(elem["levels"][idx] >= query["levels"][idx] for idx in range(6))
        )
    ]
    return filtered


def render_gem_to_str(gem_info_list: list) -> str:
    # gem info
    # {
    #     "multiplier": 2.0,
    #     "tier": 3,
    #     "count": 1
    # },
    ans = ", ".join(
        [
            f"{'*' if elem['multiplier'] == 2.0 else ''}T{elem['tier']}{'*' if elem['multiplier'] == 2.0 else ''}"
            for elem in gem_info_list
            for _ in range(elem["count"])
        ]
    )
    return ans
