def compute_damage(base_damage: float, crit_chance: float, crit_multiplier: float, super_crit_chance: float = 0):
    average_multiplier = (
        super_crit_chance * (2 + crit_chance) * crit_multiplier
        + (1 - super_crit_chance) * (
            crit_chance * 2 * crit_multiplier
            + (1 - crit_chance) * 1
        )
    )
    return base_damage * average_multiplier

def player_crit_chance() -> float:
    return input.crit_chance.get()

def player_crit_multiplier() -> float:
    bonus = 0
    return input.crit_multiplier.get()

def ability_crit_chance(is_lunar, is_barrage) -> float:
    bonus = 0
    if is_lunar:
        bonus = 100 * LUNARLIGHT_AFFINITY_CRIT_BONUS
    if is_barrage:
        bonus = 100 * FUSILLADE_CRIT_BONUS
    return (player_crit_chance() + bonus) / 100

def player_haste_percent() -> float:
    return input.haste.get() / 100


class Ability:
    base_damage: float
    name: str
    damage_multiplier: float = 1
    base_cast_time: float = 1.5
    max_num_targets: int = 1
    min_num_targets: int = 1
    secondary_target_multiplier: float = 1
    is_lunar: bool = False
    is_barrage: bool = False
    super_crit_chance: float = 0

    def __init__(
        self,
        base_damage: float,
        name: str,
        damage_multiplier: float = 1,
        base_cast_time: float = 1.5,
        max_num_targets: int = 1,
        min_num_targets: int = 1,
        secondary_target_multiplier: float = 1,
        is_lunar: bool = False,
        is_barrage: bool = False,
        super_crit_chance: float = 0,
    ):
        self.base_damage = base_damage
        self.name = name
        self.damage_multiplier = damage_multiplier
        self.base_cast_time = base_cast_time
        self.max_num_targets = max_num_targets
        self.min_num_targets = min_num_targets
        self.secondary_target_multiplier = secondary_target_multiplier
        self.is_lunar = is_lunar
        self.is_barrage = is_barrage
        self.super_crit_chance = super_crit_chance


    def total_damage(self) -> float:
        return self.damage() * self.num_hits()

    def dps(self) -> float:
        ct = self.cast_time()
        return self.total_damage() / ct if ct > 0 else float("nan")

    def cast_time(self) -> float:
        return self.base_cast_time / (1 + player_haste_percent())

    def damage(self) -> float:
        return compute_damage(
            self.base_damage * self.damage_multiplier,
            ability_crit_chance(self.is_lunar, self.is_barrage),
            player_crit_multiplier(),
            self.super_crit_chance,
        )

    def num_hits(self) -> float:
        num_secondaries = max(self.min_num_targets-1, min(self.max_num_targets-1, input.num_targets.get() - 1))
        return max(self.min_num_targets, 1 + num_secondaries * self.secondary_target_multiplier)

    def to_dict(self) -> dict[str, Any]:
        return {
            "name": self.name,
            "dps": self.dps(),
            "total_damage": self.total_damage(),
            "cast_time": self.cast_time(),
            "damage": self.damage(),
            "num_hits": self.num_hits(),
            "multiplier": self.damage_multiplier,
        }

class EmpoweredMultishot(Ability):
    base_damage: float = (MULTISHOT_DAMAGE_MIN + MULTISHOT_DAMAGE_MAX) / 2

    min_num_targets: int = 3
    max_num_targets: int = 12
    damage_bonus: float = 0.0

    def __init__(
        self,
        name: str,
        damage_bonus: float = 0.0,
    ):
        self.name = name
        self.damage_bonus = damage_bonus

    @property
    def damage_multiplier(self) -> float:
        ans = 1.0 + self.damage_bonus
        # if input.FE:  # TODO
        #     ans += 0.25
        return ans

class Barrage(Ability):
    base_damage: float = (HEARTSEEKER_BARRAGE_DAMAGE_MIN + HEARTSEEKER_BARRAGE_DAMAGE_MAX) / 2

    base_cast_time: float = 2.5
    tick_duration: float = 0.2
    damage_step: float = 0.0
    max_num_targets: int = 2
    secondary_target_multiplier: float = 0.7

    is_barrage: bool = True

    def __init__(
        self,
        name: str,
        damage_step: float = 0.0,
    ):
        self.name = name
        self.damage_step = damage_step

    def cast_time(self) -> float:
        return self.base_cast_time

    def num_ticks(self) -> int:
        return math.floor(self.base_cast_time / self.tick_duration * (1 + player_haste_percent()))

    def num_hits(self) -> float:
        num_targets = super().num_hits()
        return num_targets * self.num_ticks()

    @property
    def damage_multiplier(self) -> float:
        return (1 + (self.num_ticks() -1) * self.damage_step / 2)


class Volley(Ability):
    duration: float = 8.0
    tick_duration: float = 1
    base_damage = (VOLLEY_DAMAGE_MIN+VOLLEY_DAMAGE_MAX)/2
    max_num_targets = 12

    def __init__(self):
        self.name = "Volley"

    def num_ticks(self) -> int:
        return 1 + math.floor(self.duration / self.tick_duration * (1 + player_haste_percent()))

    def num_hits(self) -> float:
        num_targets = super().num_hits()
        return num_targets * self.num_ticks()


class AbilityAddition(Ability):

    def __init__(self, name: str, ability_list: list[Ability]):
        self.name = name
        self.ability_list = ability_list

    def dps(self):
        return self.total_damage() / self.cast_time()

    def total_damage(self):
        return sum([ability.total_damage() for ability in self.ability_list])

    def cast_time(self):
        return sum([ability.cast_time() for ability in self.ability_list])

    def damage(self):
        return float("nan")

    def num_hits(self):
        return float("nan")

    @property
    def damage_multiplier(self):
        return float("nan")
